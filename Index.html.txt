    function loadTableData() {
  setButtonLoading('#btnRefreshMain', true, 'Đang tải...');
  var tbody = document.getElementById('tblBody');
  tbody.innerHTML = '<tr><td colspan="6" style="color: var(--text-muted); padding: 30px;">Đang tải dữ liệu...</td></tr>';

  google.script.run
    .withSuccessHandler(function(response) {
      setButtonLoading('#btnRefreshMain', false);
      rawMainData = response ? (response.list || []) : [];
      filteredMainData = rawMainData.slice();
      currentMainPage = 1;
      renderMainTablePage();
    })
    .withFailureHandler(function(error) {
      // Mở khóa lại nút bấm và báo lỗi khi server gặp sự cố hoặc timeout
      setButtonLoading('#btnRefreshMain', false);
      tbody.innerHTML = '<tr><td colspan="6" style="color: #ef4444; padding: 30px;">Không thể tải dữ liệu. Vui lòng thử lại.</td></tr>';

    })
    .getRecentRowsData(currentMode);
}

    function onSearchMain() {
      var query = document.getElementById('searchNormal').value.toLowerCase().trim();
      if (!query) {
        filteredMainData = rawMainData.slice();
      } else {
        filteredMainData = rawMainData.filter(function(item) {
          var name = (item.nguoiNop || '').toLowerCase();
          var quay = (item.quay || '').toLowerCase();
          return name.indexOf(query) !== -1 || quay.indexOf(query) !== -1;
        });
      }
      currentMainPage = 1;
      renderMainTablePage();
    }

    function renderMainTablePage() {
      var tbody = document.getElementById('tblBody');
      tbody.innerHTML = '';

      if (!filteredMainData || filteredMainData.length === 0) {
        tbody.innerHTML = '<tr><td colspan="6" style="color: var(--text-muted); padding: 30px;">Không tìm thấy dữ liệu phù hợp.</td></tr>';
        document.getElementById('mainPageInfo').innerText = 'Hiển thị 0 - 0 / 0 dòng';
        document.getElementById('mainPageNum').innerText = 'Trang 1 / 1';
        document.getElementById('btnMainPrev').disabled = true;
        document.getElementById('btnMainNext').disabled = true;
        return;
      }

      var totalRows = filteredMainData.length;
      var totalPages = Math.ceil(totalRows / PAGE_SIZE);
      if (currentMainPage > totalPages) currentMainPage = totalPages;
      if (currentMainPage < 1) currentMainPage = 1;

      var startIdx = (currentMainPage - 1) * PAGE_SIZE;
      var endIdx = Math.min(startIdx + PAGE_SIZE, totalRows);
      var pageItems = filteredMainData.slice(startIdx, endIdx);

      pageItems.forEach(function(item) {
        var tr = document.createElement('tr');
        tr.innerHTML = '<td><span style="color: var(--text-muted); font-weight:700;">' + item.rowNum + '</span></td>' +
          '<td style="font-weight:600;">' + item.ngay + '</td>' +
          '<td><span class="badge-quay">' + item.quay + '</span></td>' +
          '<td style="text-align:left; font-weight: 600;">' + item.nguoiNop + '</td>' +
          '<td><span class="amount-highlight">' + (item.soTien || 0).toLocaleString('vi-VN') + '</span></td>' +
          '<td>' +
            '<div style="display: flex; gap: 8px; justify-content: center;">' +
              '<button class="btn-print" id="btnPrintRow_' + item.rowNum + '" onclick="printSingleRow(' + item.rowNum + ')">In phiếu</button>' +
              '<button class="btn-nop" id="btnNopRow_' + item.rowNum + '" onclick="executeNopRow(' + item.rowNum + ')">Nộp</button>' +
            '</div>' +
          '</td>';
        tbody.appendChild(tr);
      });

      document.getElementById('mainPageInfo').innerText = 'Hiển thị ' + (startIdx + 1) + ' - ' + endIdx + ' / ' + totalRows + ' dòng';
      document.getElementById('mainPageNum').innerText = 'Trang ' + currentMainPage + ' / ' + totalPages;
      document.getElementById('btnMainPrev').disabled = (currentMainPage === 1);
      document.getElementById('btnMainNext').disabled = (currentMainPage === totalPages);
    }

    function changeMainPage(delta) {
      currentMainPage += delta;
      renderMainTablePage();
    }

   function printSingleRow(rowNum) {
  // 1. Mở cửa sổ trống TRỰC TIẾP ngay khi người dùng click
  var win = window.open('', '_blank');
  if (!win) {
    alert("Trình duyệt đang chặn Pop-up. Vui lòng bật cho phép Pop-up cho trang web này!");
    return;
  }
  win.document.write('<html><head><title>Đang tải...</title></head><body style="font-family:sans-serif; padding:20px;"><h1>(☞ﾟヮﾟ)☞ Hệ thống đang tải... Đọc hết câu này mà chưa xong thì đọc lại lần nữa nhé. Kamsamita</h3></body></html>');

  setButtonLoading('#btnPrintRow_' + rowNum, true, 'Đang chuẩn bị in...');

  // 2. Kiểm tra tính hợp lệ của Quầy
  google.script.run.withSuccessHandler(function(check) {
    if (!check.isValid) {
      setButtonLoading('#btnPrintRow_' + rowNum, false);
      win.close(); // Đóng tab nếu thông tin quầy không hợp lệ
      alert("Ký hiệu quầy '" + check.quay + "' KHÔNG CÓ TRONG DANH SÁCH KYHIEUQUAY!\nVui lòng kiểm tra lại trước khi in.");
      return;
    }

    // 3. Lấy nội dung HTML mẫu in và ghi trực tiếp vào cửa sổ đã mở
    google.script.run.withSuccessHandler(function(html) {
      setButtonLoading('#btnPrintRow_' + rowNum, false);
      if (!html) {
        win.close();
        alert("Không lấy được dữ liệu phiếu!");
        return;
      }

      win.document.open();
      win.document.write(html);
      win.document.close();
      win.focus();
      setTimeout(function() { win.print(); }, 400);

    }).withFailureHandler(function() {
      setButtonLoading('#btnPrintRow_' + rowNum, false);
      win.close();
      alert("Có lỗi xảy ra khi tải HTML in phiếu!");
    }).getPrintHtmlByRow(rowNum, currentMode);

  }).withFailureHandler(function() {
    setButtonLoading('#btnPrintRow_' + rowNum, false);
    win.close();
    alert("Có lỗi xảy ra khi kiểm tra quầy!");
  }).checkRowQuayValid(rowNum, currentMode);
}

    function executeExportExcel() {
      var fDate = document.getElementById('fromDate').value;
      var tDate = document.getElementById('toDate').value;
      if (!fDate || !tDate) {
        alert("Vui lòng chọn từ ngày và đến ngày!");
        return;
      }
      setButtonLoading('#btnReportExcel', true, 'Đang tải Excel...');

      google.script.run.withSuccessHandler(function(data) {
        setButtonLoading('#btnReportExcel', false);
        if (!data || !data.rows || data.rows.length === 0) {
          alert("Không tìm thấy dữ liệu trong khoảng thời gian đã chọn!");
          return;
        }

        var sheetData = [];
        var titleReport = (currentMode === 'BANK') ? "BÁO CÁO TỔNG HỢP NỘP TIỀN NGÂN HÀNG" : "BÁO CÁO TỔNG HỢP NỘP TIỀN HÀNG NGÀY";
        sheetData.push([titleReport]);
        sheetData.push(["Từ ngày: " + data.fromDateDisplay + " Đến ngày: " + data.toDateDisplay]);
        sheetData.push([]);

        var headers = ["STT", "Ngày", "Quầy", "Người nộp"];
        data.fields.forEach(function(f) { headers.push(f.title); });
        headers.push("TỔNG TIỀN (VNĐ)");
        sheetData.push(headers);

        data.rows.forEach(function(item) {
          var row = [item.stt, item.ngay, item.quay, item.nguoiNop];
          data.fields.forEach(function(f, idx) {
            row.push(item.items[idx] || 0);
          });
          row.push(item.tongTien);
          sheetData.push(row);
        });

        var totalRow = ["TỔNG CỘNG", "", "", ""];
        data.fields.forEach(function(f, idx) {
          totalRow.push(data.colTotals[idx] || 0);
        });
        totalRow.push(data.grandTotal);
        sheetData.push(totalRow);

        sheetData.push([]);
        sheetData.push(["Bằng chữ:", data.grandTotalWords]);

        var wb = XLSX.utils.book_new();
        var ws = XLSX.utils.aoa_to_sheet(sheetData);
        XLSX.utils.book_append_sheet(wb, ws, "BaoCaoNopTien");

        var fileName = "BaoCao_" + (currentMode === 'BANK' ? "NganHang_" : "HangNgay_") + fDate.replace(/\//g, '') + "_" + tDate.replace(/\//g, '') + ".xlsx";
        XLSX.writeFile(wb, fileName);
        showToast("Đã xuất file Excel thành công!", true);
      }).getReportData(fDate, tDate, currentMode);
    }

          var tenQuay = (item.tenQuay || '').toLowerCase();
          return name.indexOf(query) !== -1 || quay.indexOf(query) !== -1 || tenQuay.indexOf(query) !== -1;
        });
      }
      currentDSNOPPage = 1;
      renderDSNOPTablePage();
    }

    function renderDSNOPTablePage() {
      var tbody = document.getElementById('dsnopTblBody');
      tbody.innerHTML = '';

      if (!filteredDSNOPData || filteredDSNOPData.length === 0) {
        tbody.innerHTML = '<tr><td colspan="7" style="color: var(--text-muted); padding: 30px;">Không có quầy nào phù hợp.</td></tr>';
        document.getElementById('dsnopPageInfo').innerText = 'Hiển thị 0 - 0 / 0 quầy';
        document.getElementById('dsnopPageNum').innerText = 'Trang 1 / 1';
        document.getElementById('btnDSNOPPrev').disabled = true;
        document.getElementById('btnDSNOPNext').disabled = true;
        return;
      }

      var totalRows = filteredDSNOPData.length;
      var totalPages = Math.ceil(totalRows / PAGE_SIZE);
      if (currentDSNOPPage > totalPages) currentDSNOPPage = totalPages;
      if (currentDSNOPPage < 1) currentDSNOPPage = 1;

      var startIdx = (currentDSNOPPage - 1) * PAGE_SIZE;
      var endIdx = Math.min(startIdx + PAGE_SIZE, totalRows);
      var pageItems = filteredDSNOPData.slice(startIdx, endIdx);

      pageItems.forEach(function(item, idx) {
        var tr = document.createElement('tr');
        tr.innerHTML =
          '<td><span style="color: var(--text-muted); font-weight:700;">' + (startIdx + idx + 1) + '</span></td>' +
          '<td style="font-weight:600;">' + escapeHtmlDSNOP(item.ngay) + '</td>' +
          '<td><span class="badge-quay">' + escapeHtmlDSNOP(item.quay) + '</span></td>' +
          '<td>' + escapeHtmlDSNOP(item.tenQuay) + '</td>' +
          '<td style="text-align:left; font-weight:600;">' + escapeHtmlDSNOP(item.nguoiNop) + '</td>' +
          '<td><span class="amount-highlight">' + (Number(item.tongTien) || 0).toLocaleString('vi-VN') + '</span></td>' +
          '<td>' +
            '<button class="btn-danger" id="btnDelDSNOP_' + item.rowIndex + '" onclick="deleteSingleDSNOPRow(' + item.rowIndex + ')">Xóa</button>' +
          '</td>';

        tbody.appendChild(tr);
      });

      document.getElementById('dsnopPageInfo').innerText = 'Hiển thị ' + (startIdx + 1) + ' - ' + endIdx + ' / ' + totalRows + ' quầy';
      document.getElementById('dsnopPageNum').innerText = 'Trang ' + currentDSNOPPage + ' / ' + totalPages;
      document.getElementById('btnDSNOPPrev').disabled = (currentDSNOPPage === 1);
      document.getElementById('btnDSNOPNext').disabled = (currentDSNOPPage === totalPages);
    }

    function changeDSNOPPage(delta) {
      currentDSNOPPage += delta;
      renderDSNOPTablePage();
    }

    function escapeHtmlDSNOP(value) {
      return String(value == null ? '' : value)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
    }

    function formatCurrencyInput(input) {
      var raw = input.value.replace(/\D/g, '');
      if (!raw) { input.value = ''; return; }

    }

    function parseFormattedNumber(val) {
      if (!val) return 0;
      var clean = val.toString().replace(/\D/g, '');
      return parseInt(clean, 10) || 0;
    }

    function docSoThanhChuClient(so) {
      if (so === 0) return "Không đồng";
      var chuSo = ["không", "một", "hai", "ba", "bốn", "năm", "sáu", "bảy", "tám", "chín"];
      var tien = ["", " nghìn", " triệu", " tỷ", " nghìn tỷ", " triệu tỷ"];

      function docSoHangChuc(so, dayDu) {
        var chu = "";
        var chuc = Math.floor(so / 10);
        var donVi = so % 10;
        if (chuc > 1) {
          chu += " " + chuSo[chuc] + " mươi";
          if (donVi == 1) chu += " mốt";
        } else if (chuc == 1) {
          chu += " mười";
          if (donVi == 1) chu += " một";
        } else if (dayDu && donVi > 0) {
          chu += " lẻ";
        }
        if (donVi == 5 && chuc >= 1) {
          chu += " lăm";
        } else if (donVi > 1 || (donVi == 1 && chuc == 0)) {
          chu += " " + chuSo[donVi];
        }
        return chu;
      }

      function docBlock(so, dayDu) {
        var chu = "";
        var tram = Math.floor(so / 100);
        so = so % 100;
        if (dayDu || tram > 0) {
          chu += " " + chuSo[tram] + " trăm";
          chu += docSoHangChuc(so, true);
        } else {
          chu += docSoHangChuc(so, false);
        }
        return chu;
      }

      var isNegative = so < 0;
      var strSo = Math.abs(so).toString();
      var i = strSo.length;
      var j = 0;
      var ketQua = "";
      var block = 0;

      while (i > 0) {
        block = parseInt(strSo.substring(Math.max(0, i - 3), i), 10);
        i -= 3;
        if (block > 0) {
          var chuBlock = docBlock(block, i > 0);
          ketQua = chuBlock + tien[j] + ketQua;
        }
        j++;
      }

      ketQua = ketQua.trim();
      if (ketQua.length > 0) {
        ketQua = (isNegative ? "Âm " : "") + ketQua.charAt(0).toUpperCase() + ketQua.slice(1) + " đồng chẵn./.";
      }
      return ketQua;
    }

function openSheetAtBottom() {
  var btn = document.getElementById('btnOpenSheet');
  var originalText = btn.innerText; // Lưu lại chữ gốc
  
  // Đổi trạng thái sang đang mở và khóa nút tránh bấm liên tục
  btn.innerText = "Đang mở...";
  btn.disabled = true;
  btn.style.opacity = "0.7";

  var mode = (currentMode === 'BANK') ? 'BANK' : 'DAILY';
  
  google.script.run
    .withSuccessHandler(function(url) {
      if (url) {
        window.open(url, '_blank');
      } else {
        showToast("Không lấy được đường dẫn Google Sheet!", false);
      }
      
      // Phục hồi lại nút sau 1.5 giây
      setTimeout(function() {
        btn.innerText = originalText;
        btn.disabled = false;
        btn.style.opacity = "1";
      }, 1500);
    })
    .withFailureHandler(function(error) {
      showToast("Có lỗi xảy ra khi mở sheet!", false);
      // Phục hồi lại nút nếu lỗi
      btn.innerText = originalText;
      btn.disabled = false;
      btn.style.opacity = "1";
    })
    .getSmartSheetUrl(mode);
}
